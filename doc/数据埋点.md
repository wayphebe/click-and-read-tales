```typescript
// 添加时间追踪工具函数
export class PageTimeTracker {
  private pageEnterTimes: Map<string, number> = new Map();
  
  enterPage(pageId: string): number {
    const enterTime = Date.now();
    this.pageEnterTimes.set(pageId, enterTime);
    return enterTime;
  }
  
  leavePage(pageId: string): { enterTime: number; duration: number } | null {
    const enterTime = this.pageEnterTimes.get(pageId);
    if (!enterTime) return null;
    
    const leaveTime = Date.now();
    const duration = leaveTime - enterTime;
    this.pageEnterTimes.delete(pageId);
    
    return { enterTime, duration };
  }
  
  getTimeOnPage(pageId: string): number | null {
    const enterTime = this.pageEnterTimes.get(pageId);
    if (!enterTime) return null;
    return Date.now() - enterTime;
  }
}

// 导出单例
export const pageTimeTracker = new PageTimeTracker();
```

```sql
-- 1. 页面停留时间分析
SELECT 
  page_number,
  AVG(duration_ms) as avg_duration_ms,
  PERCENTILE_CONT(0.5) WITHIN GROUP (ORDER BY duration_ms) as median_duration_ms,
  COUNT(*) as view_count
FROM user_events
WHERE event_type = 'page_leave'
  AND story_id = 'xxx'
GROUP BY page_number
ORDER BY page_number;

-- 2. 交互元素点击热力图
SELECT 
  page_number,
  element_id,
  element_position->>'x' as x,
  element_position->>'y' as y,
  COUNT(*) as click_count,
  AVG(time_since_page_enter_ms) as avg_time_to_click
FROM user_events
WHERE event_type = 'interactive_click'
  AND story_id = 'xxx'
GROUP BY page_number, element_id, element_position
ORDER BY page_number, click_count DESC;

-- 3. 答题正确率与思考时间
SELECT 
  page_number,
  COUNT(*) as total_answers,
  SUM(CASE WHEN (metadata->>'answer_correct')::boolean THEN 1 ELSE 0 END) as correct_count,
  AVG((metadata->>'thinking_time_ms')::numeric) as avg_thinking_time_ms
FROM user_events
WHERE event_type = 'question_answer'
  AND story_id = 'xxx'
GROUP BY page_number;

-- 4. 翻页方式偏好
SELECT 
  metadata->>'turn_method' as turn_method,
  COUNT(*) as count,
  ROUND(100.0 * COUNT(*) / SUM(COUNT(*)) OVER (), 2) as percentage
FROM user_events
WHERE event_type = 'page_turn'
GROUP BY metadata->>'turn_method';

-- 5. 故事完成率
SELECT 
  COUNT(DISTINCT CASE WHEN event_type = 'story_complete' THEN story_id END) as completed_stories,
  COUNT(DISTINCT CASE WHEN event_type = 'story_start' THEN story_id END) as started_stories,
  ROUND(100.0 * COUNT(DISTINCT CASE WHEN event_type = 'story_complete' THEN story_id END) / 
    NULLIF(COUNT(DISTINCT CASE WHEN event_type = 'story_start' THEN story_id END), 0), 2) as completion_rate
FROM user_events
WHERE user_id = 'xxx';
```

### 七、注意事项

1. 性能：埋点异步执行，不阻塞主流程
2. 隐私：仅记录行为数据，不记录敏感信息
3. 降级：埋点失败不影响功能
4. 数据量：考虑批量上传或本地缓存
5. 测试：在开发环境验证埋点准确性

### Bug 2: `Warning: Maximum update depth exceeded`

**问题描述**：
- 错误位置：`StoryReader.tsx:217`
- 错误原因：`useEffect` 的依赖数组中包含会在 effect 内部更新的状态，导致无限循环

**调试过程**：
1. 检查 `useEffect` 的依赖数组，发现包含 `visitedPages`、`pageInteractionCount`、`selectedAnswers`
2. 这些状态在 `useEffect` 内部或 cleanup 函数中被更新
3. 状态更新触发 effect 重新执行，形成无限循环

**解决方案**：
1. 从依赖数组中移除 `visitedPages`、`pageInteractionCount`、`selectedAnswers`
2. 使用 `useRef` 存储 `pageInteractionCount` 和 `selectedAnswers` 的最新值
3. 添加单独的 `useEffect` 同步 ref 和 state
4. 在 cleanup 函数中通过 ref 访问最新值

**关键代码修改**：
```typescript
// 使用 ref 存储最新值
const pageInteractionCountRef = useRef(0);
const selectedAnswersRef = useRef<Record<string, string>>({});

// 同步 ref 和 state
useEffect(() => {
  pageInteractionCountRef.current = pageInteractionCount;
}, [pageInteractionCount]);

useEffect(() => {
  selectedAnswersRef.current = selectedAnswers;
}, [selectedAnswers]);

// 在 cleanup 中使用 ref
return () => {
  const currentInteractionCount = pageInteractionCountRef.current;
  const currentSelectedAnswers = selectedAnswersRef.current;
  // ...
};
```

### Bug 3: 流式生成故事只显示一页，点击选项不继续生成

**问题描述**：
- 流式生成故事时，只生成了第一页
- 点击选项后，新页面没有生成或没有显示
- 控制台没有明显错误

**调试过程**：
1. 添加详细日志，追踪生成流程
2. 检查 `StreamingStoryGenerator` 是否正确生成新页面
3. 检查 `StoryReader` 是否正确同步 `currentStory` 状态
4. 发现 ID 不匹配问题：生成器的内部 ID 与数据库 UUID 不一致

**根本原因**：
1. **ID 不匹配**：`StreamingStoryGenerator` 使用前端生成的短 ID，而数据库使用 UUID
2. **状态同步失败**：`StoryReader` 中的 `useEffect` 因为 ID 不匹配，无法正确同步 `currentStory` 到本地 `storybook` 状态
3. **页面跳转失败**：即使新页面生成，`currentPageIndex` 也没有自动更新

**解决方案**：
1. **ID 同步**：在保存流式故事到数据库后，更新 store 中的 `currentStory` ID 为数据库 UUID
2. **状态同步逻辑增强**：在 `currentStory` 的 `useEffect` 中添加"情况2"，即使 ID 不完全匹配，如果检测到流式故事且页面数增加，也进行同步
3. **自动页面跳转**：当检测到新页面生成且用户在当前最后一页时，自动跳转到新页面
4. **直接页面跳转**：在 `handleAnswer` 中，直接设置 `currentPageIndex` 为新页面索引

**关键代码修改**：
```typescript
// 在 Index.tsx 中，保存后更新 ID
const savedStory = await saveStory(storyToSave, user.id);
const updatedStreamingStory: StreamingStory = {
  ...streamingStory,
  id: savedStory.id  // 使用数据库 UUID
};
updateStreamingStory(updatedStreamingStory);

// 在 StoryReader.tsx 中，增强状态同步
useEffect(() => {
  // 情况2：流式故事，页面数增加，即使 ID 不完全匹配也同步
  if (isStreamingStory && currentStory.pages.length > (storybook?.pages.length || 0)) {
    setStorybook(currentStory);
    // 更新 store 中的 ID
    updateStreamingStory({ ...currentStory, id: requestedId });
    
    // 自动跳转到新页面
    if (currentPageIndex === (storybook?.pages.length || 1) - 1) {
      setTimeout(() => {
        setCurrentPageIndex(currentStory.pages.length - 1);
      }, 300);
    }
  }
}, [currentStory, storybook, currentPageIndex]);

// 在 handleAnswer 中，直接跳转
setTimeout(() => {
  setCurrentPageIndex(updatedStory.pages.length - 1);
}, 500);
```
